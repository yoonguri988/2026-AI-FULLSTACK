    select * from mvcboard1 where bno=?
    update mvcboard1 set btitle=?, bcontent=? where bno=? and bpass=?
-5) delete.jsp (글 삭제 폼), delete_action.jsp (글 삭제 처리)
    delete from mvcboard1 where bno=? and bpass=?
